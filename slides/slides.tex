\begin{frame}{DP -- Dining Philosophers}

\textbf{Fases:}
\begin{enumerate}
  \item \textbf{\'Ordenes de pensar} --- todos los posibles \'ordenes en que piensan los $n$ fil\'osofos: $n \cdot 2^{n-1}$
  \item \textbf{Alimentar} --- alimentar a los $n$ fil\'osofos (crece con $k$): $n(k+5)+n$
  \item \textbf{Retorno} --- volver desde estado marcado (constante): $9$
\end{enumerate}

\bigskip
\textbf{F\'ormula:}
\[
  T_{\mathrm{DP}}(n,k) \;=\; \underbrace{n \cdot 2^{n-1}}_{\text{\'ordenes de pensar}}
    \;+\; \underbrace{n(k+5) + n}_{\text{alimentar}}
    \;+\; \underbrace{9}_{\text{retorno}}
\]

\end{frame}

\begin{frame}{Ahora s\'i a CM}

Cat and Mouse tiene \textbf{dos fases}:

\bigskip
\begin{description}
  \item[$A$ --- Aproximaci\'on] El rat\'on se mueve hacia el objetivo. Transiciones hasta llegar al goal.
  \item[$B$ --- Retorno] El rat\'on est\'a en el objetivo; el gato recorre sus posiciones hasta volver al inicio del ciclo.
\end{description}

\bigskip
\[
  T_{\mathrm{CM}}(n,k) = A(n,k) + B(n,k)
\]
\end{frame}

\begin{frame}{CM -- Cat and Mouse ($n=1$, simplificado)}

Para $n=1$ la f\'ormula se simplifica a:
\[
  T_{\mathrm{CM}}(1,k) = 3k^2 + 12k + 6
\]

\textbf{Dos fases estructurales:}

\medskip
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Aproximaci\'on} $A(1,k)$:
\[ A = 3k^2 - 2k + 1 \]
{\small El rat\'on toma $k$ pasos hacia el centro; el gato lo sigue desde el borde.}

\column{0.48\textwidth}
\textbf{Retorno} $B(1,k)$:
\[ B = 14k + 5 \]
{\small El rat\'on permanece en el centro; el gato recorre las $2k{+}1$ posiciones.}
\end{columns}

\bigskip
\[
  A + B = (3k^2{-}2k{+}1) + (14k{+}5) = 3k^2 + 12k + 6 \quad\checkmark
\]
\end{frame}

\begin{frame}{CM -- F\'ormula general}

\textbf{Retorno:}
\[
  B(n,k) = (2k{+}1)^{n-1} \cdot \bigl[(n{+}3)(2k{+}1) + n\cdot 2^{n-1}(6k{+}1)\bigr]
\]

\textbf{Aproximaci\'on:} \;con $f(n,i) = (n{+}1)\,i^n + n(3i{-}1)(2i{+}1)^{n-1} + (i{+}1)^n$
\[
  A(n,k) = 1 + \sum_{i=1}^{k-1} f(n,i) + n\cdot k^n, \qquad
  T_{\mathrm{CM}} = A + B
\]

\column{0.48\textwidth}
